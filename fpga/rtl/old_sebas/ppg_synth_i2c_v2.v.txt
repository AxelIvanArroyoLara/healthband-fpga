module ppg_synth #(
    parameter CLK_HZ = 50_000_000,
    parameter I2C_HZ = 400_000
)(
    input  wire        clk,
    input  wire        rst,
    output reg  [15:0] ir_sample,
    output reg  [15:0] red_sample,
    inout  wire        sda,
    inout  wire        scl
	 output wire        sample_valid
	 output wire        fifo_overflow
	 );

    localparam integer DIV = CLK_HZ / (2 * I2C_HZ);

    reg [15:0] div_cnt;
    reg i2c_tick;

    always @(posedge clk) begin
        if (rst) begin
            div_cnt  <= 0;
            i2c_tick <= 0;
        end else begin
            if (div_cnt >= DIV-1) begin
                div_cnt  <= 0;
                i2c_tick <= 1;
            end else begin
                div_cnt  <= div_cnt + 1'b1;
                i2c_tick <= 0;
            end
        end
    end

    // I2C OPEN DRAIN
    reg sda_low;
    reg scl_low;

    assign sda = sda_low ? 1'b0 : 1'bz;
    assign scl = scl_low ? 1'b0 : 1'bz;

    wire sda_in = sda;

    localparam [6:0] SENSOR_ADDR = 7'h57;

    localparam [7:0] REG_MODE_CONFIG = 8'h06;
    localparam [7:0] REG_SPO2_CONFIG = 8'h07;
    localparam [7:0] REG_LED_CONFIG  = 8'h09;
    localparam [7:0] REG_FIFO_DATA   = 8'h05;

    localparam [4:0]
        IDLE        = 0,
        START       = 1,
        DEV_ADDR    = 2,
        ACK1        = 3,
        REG_ADDR    = 4,
        ACK2        = 5,
        DATA_WRITE  = 6,
        ACK3        = 7,
        RESTART     = 8,
        DEV_READ    = 9,
        ACK4        = 10,
        DATA_READ   = 11,
        NEXT_BYTE   = 15,
        STOP        = 14;

    reg [4:0] state;
    reg [1:0] phase;
    reg [3:0] bit_cnt;

    reg [7:0] tx_data;
    reg [7:0] rx_data;

    reg [7:0] reg_addr;
    reg [7:0] write_data;

    reg [7:0] fifo_bytes [0:3];

    reg [2:0] config_index;
    reg [2:0] byte_index;

    reg configured;

    function [7:0] config_reg;
        input [2:0] index;
        begin
            case(index)
                0: config_reg = 8'h06;
                1: config_reg = 8'h07;
                2: config_reg = 8'h09;
                default: config_reg = 8'h06;
            endcase
        end
    endfunction

    function [7:0] config_value;
        input [2:0] index;
        begin
            case(index)
                0: config_value = 8'h03;
                1: config_value = 8'h47;
                2: config_value = 8'h24;
                default: config_value = 8'h00;
            endcase
        end
    endfunction

    task load_tx;
        input [7:0] value;
        begin
            tx_data <= value;
            bit_cnt <= 7;
        end
    endtask

    always @(posedge clk) begin

        if (rst) begin

            state <= IDLE;
            phase <= 0;

            sda_low <= 0;
            scl_low <= 0;

            bit_cnt <= 7;

            config_index <= 0;
            byte_index <= 0;

            configured <= 0;

            ir_sample <= 0;
            red_sample <= 0;

            tx_data <= 0;
            rx_data <= 0;

            reg_addr <= 0;
            write_data <= 0;

        end else if (i2c_tick) begin

            case(state)

                IDLE: begin
                    sda_low <= 0;
                    scl_low <= 0;
                    phase <= 0;

                    if (!configured) begin
                        reg_addr <= config_reg(config_index);
                        write_data <= config_value(config_index);
                    end else begin
                        reg_addr <= REG_FIFO_DATA;
                    end

                    state <= START;
                end

                START: begin
                    case(phase)
                        0: begin
                            scl_low <= 0;
                            sda_low <= 0;
                            phase <= 1;
                        end

                        1: begin
                            sda_low <= 1;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;
                            load_tx({SENSOR_ADDR,1'b0});
                            state <= DEV_ADDR;
                        end
                    endcase
                end

                DEV_ADDR: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= ~tx_data[bit_cnt];
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(bit_cnt == 0)
                                state <= ACK1;
                            else
                                bit_cnt <= bit_cnt - 1'b1;
                        end
                    endcase
                end

                ACK1: begin
                    case(phase)
                        0: begin
                            sda_low <= 0;
                            scl_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(sda_in == 0) begin
                                load_tx(reg_addr);
                                state <= REG_ADDR;
                            end else begin
                                state <= STOP;
                            end
                        end
                    endcase
                end

                REG_ADDR: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= ~tx_data[bit_cnt];
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(bit_cnt == 0)
                                state <= ACK2;
                            else
                                bit_cnt <= bit_cnt - 1'b1;
                        end
                    endcase
                end

                ACK2: begin
                    case(phase)
                        0: begin
                            sda_low <= 0;
                            scl_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(sda_in == 0) begin
                                if(configured)
                                    state <= RESTART;
                                else begin
                                    load_tx(write_data);
                                    state <= DATA_WRITE;
                                end
                            end else begin
                                state <= STOP;
                            end
                        end
                    endcase
                end

                DATA_WRITE: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= ~tx_data[bit_cnt];
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(bit_cnt == 0)
                                state <= ACK3;
                            else
                                bit_cnt <= bit_cnt - 1'b1;
                        end
                    endcase
                end

                ACK3: begin
                    case(phase)
                        0: begin
                            sda_low <= 0;
                            scl_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;
                            state <= STOP;
                        end
                    endcase
                end

                RESTART: begin
                    case(phase)
                        0: begin
                            sda_low <= 0;
                            scl_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            sda_low <= 1;
                            scl_low <= 0;
                            phase <= 0;
                            load_tx({SENSOR_ADDR,1'b1});
                            state <= DEV_READ;
                        end
                    endcase
                end

                DEV_READ: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= ~tx_data[bit_cnt];
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(bit_cnt == 0)
                                state <= ACK4;
                            else
                                bit_cnt <= bit_cnt - 1'b1;
                        end
                    endcase
                end

                ACK4: begin
                    case(phase)
                        0: begin
                            sda_low <= 0;
                            scl_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(sda_in == 0) begin
                                bit_cnt <= 7;
                                rx_data <= 0;
                                state <= DATA_READ;
                            end else begin
                                state <= STOP;
                            end
                        end
                    endcase
                end

                DATA_READ: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= 0;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            rx_data[bit_cnt] <= sda_in;
                            phase <= 0;

                            if(bit_cnt == 0) begin
                                fifo_bytes[byte_index] <= {
                                    rx_data[7:1],sda_in
                                };
                                state <= NEXT_BYTE;
                            end else begin
                                bit_cnt <= bit_cnt - 1'b1;
                            end
                        end
                    endcase
                end

                NEXT_BYTE: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= (byte_index == 3);
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            scl_low <= 1;
                            phase <= 0;

                            if(byte_index == 3) begin
                                state <= STOP;
                            end else begin
                                byte_index <= byte_index + 1'b1;
                                bit_cnt <= 7;
                                rx_data <= 0;
                                state <= DATA_READ;
                            end
                        end
                    endcase
                end

                STOP: begin
                    case(phase)
                        0: begin
                            scl_low <= 1;
                            sda_low <= 1;
                            phase <= 1;
                        end

                        1: begin
                            scl_low <= 0;
                            phase <= 2;
                        end

                        2: begin
                            sda_low <= 0;
                            phase <= 0;

                            if(!configured) begin
                                if(config_index == 2) begin
                                    configured <= 1;
                                    byte_index <= 0;
                                end else begin
                                    config_index <= config_index + 1'b1;
                                end
                            end else begin
                                ir_sample <= {
                                    fifo_bytes[2],fifo_bytes[3]
                                };

                                red_sample <= {
                                    fifo_bytes[0],fifo_bytes[1]
                                };

                                byte_index <= 0;
                            end

                            state <= IDLE;
                        end
                    endcase
                end

                default: begin
                    state <= IDLE;
                    phase <= 0;
                end

            endcase
        end
    end

endmodule
